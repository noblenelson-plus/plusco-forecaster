// firestore.rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // ─── Access model ──────────────────────────────────────────────────────────
    // Mirrors lib/format/access.ts — change both together.
    //   - users/{email} (lowercase): role USER | ADMIN; `disabled` revokes all.
    //   - A client's team (CL_Team_Emails on the client doc: BL, DL, GM Pod
    //     accounts, collaborators) reads + writes that client.
    //   - config/dashboard_access opens each grantable dashboard to email
    //     domains / people, in GLOBAL (every client, "_unassigned" rows
    //     included) or AGENCY mode (the agencies the caller's email domain maps
    //     to; a company-wide domain maps to all). Any grant also reads those
    //     clients' data — read-only.
    //   - Admins: everything.

    function isAuthenticated() {
      return request.auth != null;
    }
    function myEmail() {
      return request.auth.token.email.lower();
    }
    function myDomain() {
      return myEmail().split('@')[1];
    }
    function myUserPath() {
      return /databases/$(database)/documents/users/$(myEmail());
    }
    // Signed in with a verified email, has a users row, not revoked.
    function isActive() {
      return isAuthenticated() &&
        request.auth.token.email_verified == true &&
        exists(myUserPath()) &&
        get(myUserPath()).data.get('disabled', false) != true;
    }
    function isAdmin() {
      return isActive() && get(myUserPath()).data.role == 'ADMIN';
    }

    // ─── Agencies ──────────────────────────────────────────────────────────────
    function isCompanyDomain() {
      return exists(/databases/$(database)/documents/config/company_domains) &&
        myDomain() in get(/databases/$(database)/documents/config/company_domains)
          .data.get('domains', []);
    }
    function domainOwnsAgency(agency) {
      return exists(/databases/$(database)/documents/agencies/$(agency)) &&
        myDomain() in get(/databases/$(database)/documents/agencies/$(agency))
          .data.get('domains', []);
    }

    // ─── Dashboard grants (config/dashboard_access) ───────────────────────────
    function dashboards() {
      return exists(/databases/$(database)/documents/config/dashboard_access)
        ? get(/databases/$(database)/documents/config/dashboard_access).data.get('dashboards', {})
        : {};
    }
    function grantedWith(id, mode) {
      let g = dashboards().get(id, {});
      return g.get('mode', 'AGENCY') == mode &&
        (myDomain() in g.get('domains', []) || myEmail() in g.get('users', []));
    }
    function granted(id) {
      return grantedWith(id, 'GLOBAL') || grantedWith(id, 'AGENCY');
    }
    // A GLOBAL grant on any dashboard → every client and every agency row.
    function dashCanReadAll() {
      return grantedWith('labs-pacing', 'GLOBAL') || grantedWith('exec-kpis', 'GLOBAL') ||
        grantedWith('mediaocean', 'GLOBAL') || grantedWith('mediabox', 'GLOBAL') ||
        grantedWith('reports', 'GLOBAL');
    }
    // An AGENCY grant on any dashboard → the caller's agencies (all for a
    // company-wide domain), never "_unassigned".
    function dashCanReadAgency(agency) {
      return agency != '_unassigned' &&
        (grantedWith('labs-pacing', 'AGENCY') || grantedWith('exec-kpis', 'AGENCY') ||
          grantedWith('mediaocean', 'AGENCY') || grantedWith('mediabox', 'AGENCY') ||
          grantedWith('reports', 'AGENCY')) &&
        (isCompanyDomain() || domainOwnsAgency(agency));
    }

    // ─── Clients ───────────────────────────────────────────────────────────────
    function clientPath(clientId) {
      return /databases/$(database)/documents/clients/$(clientId);
    }
    function isOnClientTeam(clientId) {
      return isActive() && exists(clientPath(clientId)) &&
        myEmail() in get(clientPath(clientId)).data.get('CL_Team_Emails', []);
    }
    // Read a client's data: admin, its team, or a dashboard grant covering it.
    function canReadClientData(clientId) {
      return isAdmin() || isOnClientTeam(clientId) || (
        isActive() && exists(clientPath(clientId)) &&
        (dashCanReadAll() || dashCanReadAgency(get(clientPath(clientId)).data.CL_Agency))
      );
    }
    // Write a client's data (forecast, flags, milestones, product tracking,
    // commissions, Labs eligibility): admin or its team.
    function canWriteClient(clientId) {
      return isAdmin() || isOnClientTeam(clientId);
    }
    // Agency-partitioned dashboard data (MediaOcean collections; storage.rules
    // mirrors it for the Reports snapshots). Non-admin queries must filter
    // `_agency == <agency>` unless the caller holds a GLOBAL grant.
    function canReadAgencyData(agency) {
      return isAdmin() || (isActive() && (dashCanReadAll() || dashCanReadAgency(agency)));
    }

    // Locking lives on the rfqs document, not on data_entries.
    // data_entries carry `year` (int) and `rfq` (e.g. "RFQ0"); the matching
    // rfqs doc id is `{year}_{type}` (e.g. "2026_RFQ0").
    function rfqUnlocked(year, rfq) {
      let rfqId = string(year) + '_' + rfq;
      return exists(/databases/$(database)/documents/rfqs/$(rfqId)) &&
        get(/databases/$(database)/documents/rfqs/$(rfqId)).data.status != "LOCKED";
    }

    // ─── users ─────────────────────────────────────────────────────────────────
    // Doc id = lowercase email. Created on first sign-in (role USER) or by an
    // admin before it.
    match /users/{email} {
      // Every signed-in user reads the list (team / grant pickers, name labels).
      // Accepted trade-off: emails and roles are visible to every signed-in
      // employee — internal tool, restricted domains.
      allow read: if isAuthenticated();

      // Self sign-up: exactly the bootstrap fields, role USER, own email + uid.
      allow create: if isAdmin() || (
        isAuthenticated() &&
        request.auth.token.email_verified == true &&
        email == myEmail() &&
        request.resource.data.keys().hasOnly(['email', 'uid', 'displayName', 'photoURL',
                                               'role', 'createdBy', 'createdAt', 'lastLoginAt']) &&
        request.resource.data.email == email &&
        request.resource.data.uid == request.auth.uid &&
        request.resource.data.role == 'USER' &&
        request.resource.data.get('createdBy', null) == null
      );

      // Admins change anything. A person may refresh only their own sign-in
      // fields (linking their uid) — never role or disabled.
      allow update: if isAdmin() || (
        isAuthenticated() &&
        request.auth.token.email_verified == true &&
        email == myEmail() &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['uid', 'displayName', 'photoURL', 'lastLoginAt']) &&
        request.resource.data.uid == request.auth.uid
      );

      allow delete: if isAdmin();
    }
    // ─── clients ───────────────────────────────────────────────────────────────
    match /clients/{clientId} {
      // Admin; the team (also the shape of the "my clients" array-contains
      // query); a GLOBAL grant (whole collection); an AGENCY grant for the
      // client's agency (the `CL_Agency ==` queries). CL_Agency and
      // CL_Team_Emails are read off the document itself.
      allow read: if isAdmin() || (
        isActive() && (
          myEmail() in resource.data.CL_Team_Emails ||
          dashCanReadAll() ||
          dashCanReadAgency(resource.data.CL_Agency)
        )
      );

      // Only admins create or delete clients (and so set the team).
      allow create, delete: if isAdmin();

      // Update:
      // — Admins: no restriction (the team fields and CL_Team_Emails included).
      // — The client's team: ONLY commissionsConfig and Labs_Eligibility
      //   (+ updatedAt). Any write touching another key is denied.
      //   (CommissionsDrawer writes via the dot-path "commissionsConfig.{year}";
      //   the client drawer replaces the Labs_Eligibility map via
      //   saveClientLabsEligibility.)
      allow update: if isAdmin() ||
                    (
                      canWriteClient(clientId) &&
                      request.resource.data.diff(resource.data)
                        .affectedKeys()
                        .hasOnly(['commissionsConfig', 'Labs_Eligibility', 'updatedAt']) &&
                      (!('Labs_Eligibility' in request.resource.data) ||
                        request.resource.data.Labs_Eligibility is map)
                    );
    }
    // ─── agencies ──────────────────────────────────────────────────────────────
    // Document ID = the agency name (a ClientAgency value). Holds the email
    // domains that map to the agency (a person's agency on Agency-mode
    // dashboards). Read by every signed-in user; only admins manage it.
    match /agencies/{agencyId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isAdmin();
    }
    // ─── labs_partners ─────────────────────────────────────────────────────────
    match /labs_partners/{partnerId} {
      // All authenticated users can read partners (needed to populate LABS rows)
      allow read: if isAuthenticated();
      // Only admins can manage the global partner registry
      allow create, update, delete: if isAdmin();
    }
    // ─── resources ─────────────────────────────────────────────────────────────
    // Shared library of external links (name, description, url).
    match /resources/{resourceId} {
      // All authenticated users can read (open even on "Access pending").
      allow read: if isAuthenticated();
      // Only admins can add, edit or remove resources.
      allow create, update, delete: if isAdmin();
    }
    // ─── products (DISH Products catalog) ──────────────────────────────────────
    match /products/{productId} {
      // All authenticated users can read the catalog (needed to populate the
      // pipeline axis and the revenue Product Fees dropdown).
      allow read: if isAuthenticated();
      // Only admins can manage the global product catalog.
      allow create, update, delete: if isAdmin();
    }
    // ─── currency_rates ────────────────────────────────────────────────────────
    // Document ID format: the year as a string, e.g. "2026".
    // Holds the yearly USD → CAD conversion rate.
    match /currency_rates/{year} {
      // All authenticated users can read rates (needed wherever amounts are
      // converted between currencies)
      allow read: if isAuthenticated();
      // Only admins can set or remove conversion rates
      allow create, update, delete: if isAdmin();
    }
    // Per-year partner media-spend targets (Labs Targets admin tab). Any
    // signed-in user may read (the Executive Summary consumes them); only
    // admins may edit.
    match /partner_targets/{year} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isAdmin();
    }
    // ─── config ────────────────────────────────────────────────────────────────
    // Global app configuration docs: config/dashboard_access (dashboard grants,
    // also read by these rules), config/company_domains, config/flag_validation,
    // … Everyone reads; admins write.
    match /config/{configId} {
      allow read: if isAuthenticated();
      allow create, update, delete: if isAdmin();
    }
    // ─── data_entries ──────────────────────────────────────────────────────────
    // Document ID format: {cl_id}_{year}_{rfq} e.g. CL_001_2026_RFQ0
    // The client is the document's clientId field.
    match /data_entries/{entryId} {
      // Read: whoever may read the client's data. A non-existent doc
      // (resource == null) is a normal empty case — the dashboard reads one
      // entry per in-scope client and many have no submission yet. Allow it;
      // reading absent data leaks nothing.
      allow read: if isAuthenticated() &&
        (resource == null || canReadClientData(resource.data.clientId));

      // Create: the document is created client-side on the first Save
      // (saveAxisData → setDoc with merge), so the client's team may create it
      // — as long as the RFQ is not locked. Admins always may.
      allow create: if isAdmin() ||
        (
          canWriteClient(request.resource.data.clientId) &&
          rfqUnlocked(request.resource.data.year, request.resource.data.rfq)
        ) ||
        // Submission annotations (shared note, flag reviews) — the first write
        // may annotate a submission that has no grid data yet. Allowed for the
        // team even when the RFQ is locked (annotation-only create: nothing but
        // these metadata keys is written).
        (
          canWriteClient(request.resource.data.clientId) &&
          request.resource.data.keys()
            .hasOnly(['clientId', 'year', 'rfq', 'submissionNote',
                      'flagReviews', 'flags', 'createdAt', 'lastModifiedBy',
                      'updatedAt'])
        );

      // Update: admin always — the team while the RFQ is unlocked, and without
      // re-keying the document (clientId stays put).
      allow update: if isAdmin() ||
        (
          canWriteClient(resource.data.clientId) &&
          request.resource.data.clientId == resource.data.clientId &&
          rfqUnlocked(resource.data.year, resource.data.rfq)
        ) ||
        // Submission annotations (shared note, flag reviews) — the team may
        // add/edit these even when the RFQ is locked. The write only touches
        // annotation keys (+ updatedAt); every forecast field keeps its value,
        // so this can never alter locked data.
        (
          canWriteClient(resource.data.clientId) &&
          request.resource.data.clientId == resource.data.clientId &&
          request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['submissionNote', 'flagReviews', 'flags',
                      'lastModifiedBy', 'updatedAt'])
        );

      // Actuals (ADMIN_INPUT) and closed months are enforced by the app: the
      // rules cannot easily diff nested map fields.
      // Delete: admins only — used by the client-deletion cleanup
      // (deleteClient in client-service.ts).
      allow delete: if isAdmin();
    }
    // ─── forecast_validations (Milestones) ─────────────────────────────────────
    // Document ID format: {cl_id}_{year} e.g. CL_001_2026
    // The milestone steps (confirmation-steps.ts) the team ticks as complete,
    // at the {client, year} level. Purely indicative (no locking effect): the
    // team (or an admin) may write anytime. Only these annotation keys are
    // ever written.
    match /forecast_validations/{entryId} {
      // An absent doc is a normal empty case (mirrors data_entries).
      allow read: if isAuthenticated() &&
        (resource == null || canReadClientData(resource.data.clientId));

      allow create: if isAdmin() ||
        (
          canWriteClient(request.resource.data.clientId) &&
          request.resource.data.keys()
            .hasOnly(['clientId', 'year', 'steps', 'stepValidations', 'meta',
                      'createdAt', 'updatedAt'])
        );
      allow update: if isAdmin() ||
        (
          canWriteClient(resource.data.clientId) &&
          request.resource.data.clientId == resource.data.clientId &&
          request.resource.data.diff(resource.data).affectedKeys()
            .hasOnly(['steps', 'stepValidations', 'meta', 'createdAt', 'updatedAt'])
        );

      // Delete: admins only — client-deletion cleanup (deleteClient).
      allow delete: if isAdmin();
    }
    // ─── annual_actuals ────────────────────────────────────────────────────────
    // Document ID format: {cl_id}_{year} e.g. CL_001_2026
    // The annual ADMIN_INPUT (MediaOcean) for Media and Labs — admin-owned.
    match /annual_actuals/{entryId} {
      // An absent doc is the common case (rarely populated). Allow it.
      allow read: if isAuthenticated() &&
        (resource == null || canReadClientData(resource.data.clientId));
      // Create/update: admins only (the team never edits actuals). RFQ locking
      // does not apply — these figures are not tied to a single submission.
      allow create, update: if isAdmin();
      // Delete: admins only — client-deletion cleanup (deleteClient).
      allow delete: if isAdmin();
    }
    // ─── mediabox_totals ───────────────────────────────────────────────────────
    // Document ID format: {cl_id}_{year} e.g. CL_001_2026
    // Read-only MediaBox actuals, synced by the MediaBox project's Cloud
    // Function (Admin SDK, bypasses these rules). An absent doc is the normal
    // pre-sync case — allow it like annual_actuals.
    match /mediabox_totals/{entryId} {
      allow read: if isAuthenticated() &&
        (resource == null || canReadClientData(resource.data.clientId));
      // Never writable from a client. Admins may delete (client-deletion
      // cleanup).
      allow create, update: if false;
      allow delete: if isAdmin();
    }
    // ─── mediabox_coverage ─────────────────────────────────────────────────────
    // Document ID format: {year} e.g. 2026. One doc per year holding EVERY
    // client's MediaBox totals. Because it spans all clients it is admin-only;
    // the dashboard reads the per-client mediabox_totals docs above instead.
    match /mediabox_coverage/{year} {
      allow read: if isAdmin();
      allow create, update: if false;
      allow delete: if isAdmin();
    }
    // ─── product_tracking ──────────────────────────────────────────────────────
    // Document ID = the client's cl_id (one always-on doc per client, no
    // year/RFQ). Not tied to a submission, so RFQ locking does not apply: the
    // team (or an admin) may write at any time.
    match /product_tracking/{clientId} {
      allow read: if canReadClientData(clientId);
      // The doc must stay keyed to its client (clientId field matches the id).
      allow create, update: if request.resource.data.clientId == clientId &&
        canWriteClient(clientId);
      // Delete: admins only — client-deletion cleanup (deleteClient).
      allow delete: if isAdmin();
    }
    // ─── rfqs ──────────────────────────────────────────────────────────────────
    // Document ID format: {year}_{type} e.g. 2026_RFQ1
    // May carry `closedMonths.{media|revenue|labs}: number[]` — admin-set
    // per-axis closed periods (an app-side editing gate for non-admins; not
    // enforced here, like the global lock above which only checks `status`).
    match /rfqs/{rfqId} {
      allow read: if isAuthenticated();
      // Only admins create, lock/unlock, set closed months, edit timelines or
      // delete RFQs.
      allow create, update, delete: if isAdmin();
    }
    // ─── MediaOcean dashboard collections (agency-partitioned) ────────────────
    // Synced read-only from BigQuery by scripts/sync-*.mjs (Admin SDK, which
    // bypasses these rules). Each doc carries `_agency` (scripts/lib/agency.mjs);
    // reads are scoped by canReadAgencyData. Never writable from the app;
    // admins may delete.
    // mo_kpi_by_client — one doc per client (scripts/sync-kpi-by-client.mjs).
    // Listed by agency (canReadAgencyData); a single doc may also be read by
    // the client's team — the client-scoped dashboards (Labs Pacing, Exec KPI,
    // MediaBox) fetch the allocated clients' docs one by one.
    match /mo_kpi_by_client/{clientId} {
      allow list: if canReadAgencyData(resource.data._agency);
      allow get: if canReadAgencyData(resource.data._agency) || isOnClientTeam(clientId);
      allow create, update: if false;
      allow delete: if isAdmin();
    }
    // mediaocean_investment_mix — client × month × channel × partner grain.
    match /mediaocean_investment_mix/{docId} {
      allow read: if canReadAgencyData(resource.data._agency);
      allow create, update: if false;
      allow delete: if isAdmin();
    }
    // social_partner_mix — client × partner × month grain.
    match /social_partner_mix/{docId} {
      allow read: if canReadAgencyData(resource.data._agency);
      allow create, update: if false;
      allow delete: if isAdmin();
    }
    // meta_social_output — one doc per client; read by the Exec KPI Dashboard
    // only, and not agency-tagged: admins and anyone granted that dashboard.
    match /meta_social_output/{docId} {
      allow read: if isAdmin() || (isActive() && granted('exec-kpis'));
      allow create, update: if false;
      allow delete: if isAdmin();
    }
    // dashboard_meta — the "Last Updated" stamp written by scripts/sync-all.mjs.
    // A date label only, no client data: any signed-in user may read it.
    match /dashboard_meta/{docId} {
      allow read: if isAuthenticated();
      allow create, update: if false;
      allow delete: if isAdmin();
    }
    // ─── bulk_backups ──────────────────────────────────────────────────────────
    // Automatic backups taken by Bulk Edit before every import / bulk delete
    // (bulk-backup-service.ts). They hold raw forecast data of every client, so
    // they are admin-only end to end. (auth_migration_backups, written by
    // scripts/migrate-auth.mjs, has no block: client SDKs are denied.)
    match /bulk_backups/{backupId} {
      allow read, create, update, delete: if isAdmin();
      match /snapshots/{snapId} {
        allow read, create, update, delete: if isAdmin();
      }
    }
    // mir_raw / billing_summary_raw — legacy raw mirrors (no longer written;
    // the app reads the per-agency Storage snapshots instead).
    match /mir_raw/{docId} {
      allow read: if isAdmin();
      allow create, update: if false;
      allow delete: if isAdmin();
    }
    match /billing_summary_raw/{docId} {
      allow read: if isAdmin();
      allow create, update: if false;
      allow delete: if isAdmin();
    }
  }
}
